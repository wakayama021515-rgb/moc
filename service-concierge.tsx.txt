const ServiceConcierge = () => {
  const [input, setInput] = useState('');
  const [perspectives, setPerspectives] = useState([]);
  const [expandedPerspectives, setExpandedPerspectives] = useState(new Set());
  const [selectedService, setSelectedService] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  // Claude APIを使った動的推論
  const analyzeWithClaude = async (userInput) => {
    setLoading(true);
    setError(null);

    try {
      const prompt = `あなたは社内サービスコンシェルジュのAIです。ユーザーの発話から、適切な観点（軸）を生成し、各観点に対して関連するサービスを確率的にスコアリングしてください。

【ユーザーの発話】
"${userInput}"

【利用可能な社内サービス一覧】
${SERVICES.map(s => `${s.id}. ${s.name} (${s.category}) - ${s.description}`).join('\n')}

【タスク】
1. ユーザーの発話から状況・意図・背景を推測
2. 3〜5個の「観点（perspective）」を生成（例：「受け入れ準備」「オンボーディング」など）
3. 各観点について、関連度の高いサービスを5〜8個選び、確率的スコア（0.0〜1.0）を付与

【重要】
- 観点名は具体的で行動可能なものにする
- スコアは、その観点においてそのサービスがどれだけ重要か/関連するかを示す
- 各観点には priority（0.0〜1.0）も付与（その観点自体がこの状況でどれだけ重要か）

【出力形式】
必ず以下のJSON形式のみで回答してください。他のテキストは一切含めないでください：

{
  "perspectives": [
    {
      "name": "観点名",
      "priority": 0.92,
      "services": [
        {
          "serviceId": 1,
          "score": 0.89,
          "reason": "この観点で重要な理由の簡潔な説明"
        }
      ]
    }
  ]
}

DO NOT OUTPUT ANYTHING OTHER THAN VALID JSON. レスポンスはJSONのみです。`;

      const response = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          model: "claude-sonnet-4-20250514",
          max_tokens: 4000,
          messages: [
            { role: "user", content: prompt }
          ]
        })
      });

      if (!response.ok) {
        throw new Error(`API request failed: ${response.status}`);
      }

      const data = await response.json();
      let responseText = data.content[0].text;
      
      // JSONマークダウンを除去
      responseText = responseText.replace(/```json\n?/g, "").replace(/```\n?/g, "").trim();
      
      const result = JSON.parse(responseText);

      // サービスIDを実際のサービスオブジェクトにマッピング
      const enrichedPerspectives = result.perspectives.map(persp => ({
        ...persp,
        services: persp.services.map(svc => {
          const service = SERVICES.find(s => s.id === svc.serviceId);
          return {
            service,
            score: svc.score,
            reason: svc.reason
          };
        }).filter(svc => svc.service) // 存在しないIDを除外
      }));

      setPerspectives(enrichedPerspectives);
      
      // 上位2つの観点を自動展開
      const topPerspectives = new Set(enrichedPerspectives.slice(0, 2).map(p => p.name));
      setExpandedPerspectives(topPerspectives);

    } catch (err) {
      console.error('Analysis error:', err);
      setError(`エラーが発生しました: ${err.message}`);
    } finally {
      setLoading(false);
    }
  };

  const handleSearch = () => {
    if (input.trim()) {
      analyzeWithClaude(input);
    }
  };

  const togglePerspective = (perspectiveName) => {
    const newExpanded = new Set(expandedPerspectives);
    if (newExpanded.has(perspectiveName)) {
      newExpanded.delete(perspectiveName);
    } else {
      newExpanded.add(perspectiveName);
    }
    setExpandedPerspectives(newExpanded);
  };

  return (
    <div className="min-h-screen bg-gradient-to-br from-blue-50 to-indigo-50 p-4">
      <div className="max-w-6xl mx-auto">
        {/* ヘッダー */}
        <div className="text-center mb-8 pt-8">
          <h1 className="text-4xl font-bold text-gray-800 mb-2">社内サービスコンシェルジュ</h1>
          <p className="text-gray-600">AIがあなたの状況から最適なサービスを提案します</p>
          <div className="mt-2 px-3 py-1 bg-green-100 text-green-700 rounded-full text-sm inline-block">
            ✨ AI推論エンジン搭載
          </div>
        </div>

        {/* 検索ボックス */}
        <div className="bg-white rounded-lg shadow-lg p-6 mb-8">
          <div className="flex gap-3">
            <input
              type="text"
              value={input}
              onChange={(e) => setInput(e.target.value)}
              onKeyPress={(e) => e.key === 'Enter' && handleSearch()}
              placeholder="例: 来月から新メンバーが2人入る、出張に行く、体調が悪い..."
              className="flex-1 px-4 py-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
              disabled={loading}
            />
            <button
              onClick={handleSearch}
              disabled={loading}
              className="px-6 py-3 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition flex items-center gap-2 disabled:bg-gray-400"
            >
              {loading ? <Loader size={20} className="animate-spin" /> : <Search size={20} />}
              {loading ? 'AI分析中...' : '探す'}
            </button>
          </div>
        </div>

        {/* エラー表示 */}
        {error && (
          <div className="bg-red-50 border border-red-200 rounded-lg p-4 mb-8">
            <p className="text-red-700">{error}</p>
          </div>
        )}

        {/* ローディング表示 */}
        {loading && (
          <div className="text-center py-12">
            <Loader size={48} className="mx-auto animate-spin text-blue-600 mb-4" />
            <p className="text-gray-600">AIが状況を分析しています...</p>
            <p className="text-sm text-gray-500 mt-2">観点を生成し、最適なサービスをスコアリング中</p>
          </div>
        )}

        {/* 観点とサービス一覧 */}
        {!loading && perspectives.length > 0 && (
          <div className="space-y-4">
            {perspectives.map((perspective) => (
              <div key={perspective.name} className="bg-white rounded-lg shadow-md overflow-hidden">
                {/* 観点ヘッダー */}
                <div
                  onClick={() => togglePerspective(perspective.name)}
                  className="flex items-center justify-between p-5 cursor-pointer hover:bg-gray-50 transition"
                >
                  <div className="flex items-center gap-3">
                    <h2 className="text-xl font-semibold text-gray-800">{perspective.name}</h2>
                    <div className="px-3 py-1 bg-blue-100 text-blue-700 rounded-full text-sm font-medium">
                      重要度 {Math.round(perspective.priority * 100)}%
                    </div>
                  </div>
                  {expandedPerspectives.has(perspective.name) ? (
                    <ChevronUp className="text-gray-400" />
                  ) : (
                    <ChevronDown className="text-gray-400" />
                  )}
                </div>

                {/* サービスリスト */}
                {expandedPerspectives.has(perspective.name) && (
                  <div className="border-t border-gray-100 p-5">
                    <div className="space-y-3">
                      {perspective.services.map(({ service, score, reason }) => (
                        <div
                          key={service.id}
                          onClick={() => setSelectedService(service)}
                          className="flex items-start gap-4 p-4 rounded-lg border border-gray-200 hover:border-blue-300 hover:bg-blue-50 transition cursor-pointer"
                        >
                          <div className="flex-1">
                            <div className="flex items-center gap-2 mb-1">
                              <h3 className="font-medium text-gray-800">{service.name}</h3>
                              <span className="text-xs px-2 py-1 bg-gray-100 text-gray-600 rounded">
                                {service.category}
                              </span>
                            </div>
                            <p className="text-sm text-gray-600 mb-2">{service.description}</p>
                            {reason && (
                              <p className="text-xs text-blue-600 bg-blue-50 px-2 py-1 rounded inline-block">
                                💡 {reason}
                              </p>
                            )}
                            <div className="flex gap-1 mt-2">
                              {service.tags.slice(0, 3).map(tag => (
                                <span key={tag} className="text-xs px-2 py-1 bg-gray-50 text-gray-600 rounded">
                                  {tag}
                                </span>
                              ))}
                            </div>
                          </div>
                          <div className="flex flex-col items-end gap-2">
                            <div className="text-sm font-semibold text-blue-600">
                              {Math.round(score * 100)}%
                            </div>
                            <div className="w-24 h-2 bg-gray-200 rounded-full overflow-hidden">
                              <div
                                className="h-full bg-gradient-to-r from-blue-500 to-purple-500 rounded-full"
                                style={{ width: `${score * 100}%` }}
                              />
                            </div>
                          </div>
                        </div>
                      ))}
                    </div>
                  </div>
                )}
              </div>
            ))}
          </div>
        )}

        {/* サービス詳細モーダル */}
        {selectedService && (
          <div
            className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4 z-50"
            onClick={() => setSelectedService(null)}
          >
            <div
              className="bg-white rounded-lg max-w-2xl w-full p-6"
              onClick={(e) => e.stopPropagation()}
            >
              <div className="flex items-start justify-between mb-4">
                <div>
                  <h2 className="text-2xl font-bold text-gray-800 mb-2">{selectedService.name}</h2>
                  <span className="px-3 py-1 bg-blue-100 text-blue-700 rounded-full text-sm">
                    {selectedService.category}
                  </span>
                </div>
                <button
                  onClick={() => setSelectedService(null)}
                  className="text-gray-400 hover:text-gray-600 text-2xl"
                >
                  ✕
                </button>
              </div>
              <p className="text-gray-700 mb-4">{selectedService.description}</p>
              <div className="mb-4">
                <h3 className="font-semibold text-gray-700 mb-2">機能タグ</h3>
                <div className="flex flex-wrap gap-2">
                  {selectedService.tags.map(tag => (
                    <span key={tag} className="px-3 py-1 bg-blue-50 text-blue-600 rounded-full text-sm">
                      {tag}
                    </span>
                  ))}
                </div>
              </div>
            </div>
          </div>
        )}

        {/* 初期状態のプレースホルダー */}
        {perspectives.length === 0 && input === '' && !loading && (
          <div className="text-center py-12">
            <div className="text-gray-400 mb-4">
              <Info size={48} className="mx-auto" />
            </div>
            <p className="text-gray-600 mb-4">どんな状況ですか？AIが最適なサービスを提案します</p>
            <div className="flex flex-wrap justify-center gap-2">
              {['新メンバーが入る', '出張に行く', 'プロジェクト開始', '体調不良', 'データ分析したい'].map(example => (
                <button
                  key={example}
                  onClick={() => {
                    setInput(example);
                    setTimeout(() => analyzeWithClaude(example), 100);
                  }}
                  className="px-4 py-2 bg-gray-100 text-gray-700 rounded-full hover:bg-gray-200 transition text-sm"
                >
                  {example}
                </button>
              ))}
            </div>
          </div>
        )}
      </div>
    </div>
  );
};

export default ServiceConcierge;
